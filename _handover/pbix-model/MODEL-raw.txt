    SUMX(
        CALCULATETABLE(
            DISTINCT('الرولينج'[بورصة الرولينج]),
            FILTER(
                ALL('الرولينج'),
                'الرولينج'[الرقم الداخلي] = CurrentInternalNo && 'الرولينج'[الصنف] = CurrentCategory
            )
        ),
        VAR CurrentBourse = 'الرولينج'[بورصة الرولينج]
        RETURN
        CALCULATE(
            DIVIDE(
                SUMX('الرولينج', 'الرولينج'[كمية الترويل] * 'الرولينج'[سعر الرولينج]),
                SUM('الرولينج'[كمية الترويل]),
                0
            ),
            FILTER(
                ALL('الرولينج'),
                'الرولينج'[الرقم الداخلي] = CurrentInternalNo && 
                'الرولينج'[الصنف] = CurrentCategory && 
                'الرولينج'[بورصة الرولينج] = CurrentBourse
            )
        )
    )

RETURN
BasePremium + COALESCE(RollingCumulativeImpact, 0)
